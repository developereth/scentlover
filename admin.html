`; return new Response(html, { headers: { 'Content-Type': 'text/html' } }); } if (path === '/api/products' && method === 'GET') { const { results } = await env.DB.prepare('SELECT * FROM products ORDER BY created_at DESC').all(); return new Response(JSON.stringify({ success: true, products: results }), { headers: h }); } if (path === '/api/products' && method === 'POST') { const body = await request.json(); await env.DB.prepare('INSERT INTO products (name, brand, price, category, image, description, tag) VALUES (?, ?, ?, ?, ?, ?, ?)').bind(body.name, body.brand, body.price, body.category, body.image||'', body.description||'', body.tag||'').run(); return new Response(JSON.stringify({ success: true }), { headers: h }); } if (method === 'DELETE') { const id = path.split('/api/products/')[1]; if (id) { await env.DB.prepare('DELETE FROM products WHERE id=?').bind(id).run(); return new Response(JSON.stringify({ success: true }), { headers: h }); } } if (method === 'PUT') { const id = path.split('/api/products/')[1]; if (id) { const body = await request.json(); await env.DB.prepare('UPDATE products SET name=?, brand=?, price=?, category=?, image=?, description=?, tag=?
